JavaScript Function Declaration

Example 1 - Display Message

function showMessage(){
    document.getElementById("output1").textContent =
        "Hello JavaScript!";
}

Example 2 - Welcome User

function welcome(){
    document.getElementById("output2").textContent =
        "Welcome Student!";
}

Example 3 - Greeting Function

function greet(name){
    document.getElementById("output3").textContent =
        "Hello " + name;
}

Example 4 - Add Two Numbers

function add(a,b){
    return a+b;
}

Example 5 - Multiply Two Numbers

function multiply(a,b){
    return a*b;
}
