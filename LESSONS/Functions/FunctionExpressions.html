JavaScript Function Expressions

Example 1 - Basic Function Expression

const showMessage = function () {
    return "Hello JavaScript!";
};

Example 2 - Function Expression Without Parameter

const welcome = function () {
    return "Welcome Student!";
};

Example 3 - Function Expression With Parameter

const greet = function(name){
    return "Hello " + name;
};

Example 4 - Two Parameters

const add = function(a,b){
    return a + b;
};

Example 5 - Returning Value

const multiply = function(a,b){
    return a * b;
};

Example 6 - Anonymous Function Expression

const sayHello = function(){
    return "Hello World!";
};

Example 7 - Named Function Expression

const message = function greet(){
    return "Good Morning!";
};

Example 8 - Calculate Square

const square = function(number){
    return number * number;
};

Example 9 - Full Name

const fullName = function(first,last){
    return first + " " + last;
};
